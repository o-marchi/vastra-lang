(fn row [screen t y x right w0 w1 w2]
  (if (> (to-float x) right)
    screen
    (do
      (let v0 (:v0 t))
      (let v1 (:v1 t))
      (let v2 (:v2 t))
      (let screen (if (or (< w0 0.0) (< w1 0.0) (< w2 0.0))
                    screen
                    (do
                      (let s (+ (* w0 (:iz v0)) (* w1 (:iz v1)) (* w2 (:iz v2))))
                      (plot screen (+ (* y width) x) (* s (:per t)) (:color t)))))
      (row screen t y (inc x) right
           (+ w0 (- (:sy v1) (:sy v2)))
           (+ w1 (- (:sy v2) (:sy v0)))
           (+ w2 (- (:sy v0) (:sy v1)))))))

(fn rows [screen t y bottom left right]
  (if (> (to-float y) bottom)
    screen
    (do
      (let px (+ left 0.5))
      (let py (+ (to-float y) 0.5))
      (let v0 (:v0 t))
      (let v1 (:v1 t))
      (let v2 (:v2 t))
      (let screen (row screen t y (to-int left) right
                       (edge v1 v2 px py) (edge v2 v0 px py) (edge v0 v1 px py)))
      (rows screen t (inc y) bottom left right))))

(fn fill [screen v0 v1 v2 color]
  (let area (edge v0 v1 (:sx v2) (:sy v2)))
  (let left (max 0.0 (floor (min (:sx v0) (:sx v1) (:sx v2)))))
  (let right (min (to-float (- width 1)) (ceil (max (:sx v0) (:sx v1) (:sx v2)))))

    screen
    (rows screen {:v0 v0 :v1 v1 :v2 v2 :color color :per (/ 1.0 area)}
          (to-int top) bottom left right)))

;;;; file: stopwatch.vas
(ns stopwatch (:require [vastra.clock :as clock] [vastra.console :as console]))
; A stopwatch, driven by messages sent from the file.
(type Msg Start Pause (Tick Float))
(let shown 0.0)
(let ticks 0)
(let ticking task/ended)

(fn every! [seconds wrap]          ; a repeating timer is Vastra code
  @(clock/sleep seconds)
  (send! (wrap @clock/elapsed))
  @(every! seconds wrap))

(on Start
  (stop! ticking)                  ; never two timers at once
  (set! ticking (later (every! 0.1 Tick))))
(on Pause (stop! ticking))
(on (Tick at)
  (set! shown at)
  (update! ticks inc))

(send! Start)
@(clock/sleep 0.35)
(send! Pause)
@(clock/sleep 0.01)
(let paused ticks)
@(clock/sleep 0.3)
@(console/print (str "ticked: " (> paused 0) ", none after the pause: " (= ticks paused)
                     ", shown at a tick: " (> shown 0.0)))

;;; run: a-window-runs-frames-until-it-closes
window.vas
;;;; file: window.vas
; The frames run once the file has: each goes to the handlers, then the view
; is drawn, with too few pixels, which is fine. Closing the window ends them.
(ns app (:require [vastra.window :as window] [vastra.console :as console]))
(let frames 0)
(let screen @(window/open {:title "frames" :width 4 :height 3 :scale 2}))
(on (window/Frame _ _)
  (update! frames inc)
  @(console/print (str "frame " frames))
  (if (= frames 3)
    (do @(console/print "closing") @(window/close screen))
    unit))
(view screen [0xff0000ff])
